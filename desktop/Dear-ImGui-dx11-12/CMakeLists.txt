cmake_minimum_required(VERSION 3.29)
project(NeutronStarImGui VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_LINKER_TYPE MSVC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT IMGUI_SOURCE)
 message(FATAL_ERROR "Run Init.cmd, then Build.ps1.")
endif()
file(TO_CMAKE_PATH "${IMGUI_SOURCE}" IMGUI_SOURCE)
add_library(imgui STATIC "${IMGUI_SOURCE}/imgui.cpp" "${IMGUI_SOURCE}/imgui_draw.cpp" "${IMGUI_SOURCE}/imgui_tables.cpp" "${IMGUI_SOURCE}/imgui_widgets.cpp"
 "${IMGUI_SOURCE}/backends/imgui_impl_win32.cpp" "${IMGUI_SOURCE}/backends/imgui_impl_dx11.cpp" "${IMGUI_SOURCE}/backends/imgui_impl_dx12.cpp")
target_include_directories(imgui SYSTEM PUBLIC "${IMGUI_SOURCE}" "${IMGUI_SOURCE}/backends")
target_compile_definitions(imgui PUBLIC IMGUI_DISABLE_OBSOLETE_FUNCTIONS IMGUI_USE_WCHAR32)
target_compile_options(imgui PRIVATE /utf-8 /guard:cf)
target_link_libraries(imgui PUBLIC d3d11 d3d12 dxgi d3dcompiler dwmapi imm32)
function(add_app target)
 add_executable(${target} WIN32 src/main.cpp src/ui.cpp src/graphics.cpp src/app.manifest)
 target_link_libraries(${target} PRIVATE imgui shell32)
 target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
 target_link_options(${target} PRIVATE /GUARD:CF /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /CETCOMPAT)
endfunction()
add_app(neutronstar_imgui)
add_app(neutronstar_imgui_probe)
target_compile_definitions(neutronstar_imgui_probe PRIVATE NEUTRONSTAR_PROBE)
enable_testing()
add_executable(domain_tests tests/domain_tests.cpp)
target_compile_options(domain_tests PRIVATE /W4 /WX /permissive- /utf-8)
add_test(NAME imgui-domain COMMAND domain_tests)
install(TARGETS neutronstar_imgui RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES "${IMGUI_SOURCE}/LICENSE.txt" DESTINATION Notices RENAME Dear-ImGui-MIT.txt)
