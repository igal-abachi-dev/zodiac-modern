cmake_minimum_required(VERSION 3.29)
project(NeutronStarWx VERSION 1.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_LINKER_TYPE MSVC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
set(wxBUILD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE STRING "" FORCE)
set(wxBUILD_TESTS OFF CACHE STRING "" FORCE)
set(wxBUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(wxBUILD_PRECOMP OFF CACHE STRING "" FORCE)
set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
set(wxUSE_STL ON CACHE BOOL "" FORCE)
set(wxUSE_UNSAFE_WXSTRING_CONV OFF CACHE BOOL "" FORCE)
set(wxUSE_DPI_AWARE_MANIFEST "per-monitor" CACHE STRING "" FORCE)
if(NOT WX_SOURCE)
  message(FATAL_ERROR "Run Init.cmd first, then Build.ps1.")
endif()
add_subdirectory("${WX_SOURCE}" wx EXCLUDE_FROM_ALL SYSTEM)
add_executable(neutronstar_wx WIN32 src/main.cpp)
target_link_libraries(neutronstar_wx PRIVATE wx::core wx::base gdi32)
target_compile_options(neutronstar_wx PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
target_link_options(neutronstar_wx PRIVATE /GUARD:CF /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /CETCOMPAT)
target_sources(neutronstar_wx PRIVATE src/app.manifest)
enable_testing()
add_executable(domain_tests tests/domain_tests.cpp)
target_compile_options(domain_tests PRIVATE /W4 /WX /permissive- /utf-8)
add_test(NAME wx-domain COMMAND domain_tests)
add_executable(rasterize_icons EXCLUDE_FROM_ALL tools/rasterize_icons.cpp)
target_link_libraries(rasterize_icons PRIVATE wx::core wx::base)
install(TARGETS neutronstar_wx RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
