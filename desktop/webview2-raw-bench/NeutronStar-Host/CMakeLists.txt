cmake_minimum_required(VERSION 3.25)
project(NeutronStarHost VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use clang-cl (default) or MSVC with the Microsoft Windows SDK/ABI.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NEUTRONSTAR_STATIC_CRT "Use /MT CRT (consumer targets must match)" ${PROJECT_IS_TOP_LEVEL})
if(NEUTRONSTAR_STATIC_CRT)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(WEBVIEW2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.packages/microsoft.web.webview2.1.0.4258.31" CACHE PATH "Verified WebView2 SDK")
set(WIL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.packages/microsoft.windows.implementationlibrary.1.0.260126.7" CACHE PATH "Verified WIL SDK")
foreach(header "${WEBVIEW2_ROOT}/build/native/include/WebView2.h" "${WIL_ROOT}/include/wil/com.h")
  if(NOT EXISTS "${header}")
    message(FATAL_ERROR "Run scripts/Restore-Dependencies.ps1 first. Missing ${header}")
  endif()
endforeach()
if(CMAKE_GENERATOR_PLATFORM MATCHES "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|aarch64")
  set(NS_ARCH arm64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(NS_ARCH x64)
else()
  message(FATAL_ERROR "Only x64 and ARM64 are supported.")
endif()
add_library(neutronstar_host STATIC src/host_window.cpp src/asset_policy.cpp src/test_greet_protocol.cpp)
add_library(NeutronStar::Host ALIAS neutronstar_host)
target_compile_features(neutronstar_host PUBLIC cxx_std_20)
target_include_directories(neutronstar_host PUBLIC include PRIVATE src)
target_include_directories(neutronstar_host SYSTEM PRIVATE "${WEBVIEW2_ROOT}/build/native/include" "${WIL_ROOT}/include")
target_compile_definitions(neutronstar_host PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX PRIVATE _WIN32_WINNT=0x0A00)
target_compile_options(neutronstar_host PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /guard:cf)
target_link_libraries(neutronstar_host PRIVATE "${WEBVIEW2_ROOT}/build/native/${NS_ARCH}/WebView2LoaderStatic.lib" user32 gdi32 ole32 oleaut32 shlwapi version)
add_executable(neutronstar_host_runner WIN32 examples/runner.cpp examples/app.rc)
target_link_libraries(neutronstar_host_runner PRIVATE NeutronStar::Host shell32)
target_include_directories(neutronstar_host_runner PRIVATE src)
target_compile_options(neutronstar_host_runner PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /guard:cf)
# app.rc embeds the explicit asInvoker/PerMonitorV2 manifest; suppress the generated duplicate.
target_link_options(neutronstar_host_runner PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
set(NEUTRONSTAR_EMBED_ASSETS "" CACHE PATH "Caller-supplied trusted production asset tree; empty selects file-only tooling")
if(NEUTRONSTAR_EMBED_ASSETS)
  get_filename_component(NS_EMBED_ROOT "${NEUTRONSTAR_EMBED_ASSETS}" ABSOLUTE)
  if(NOT EXISTS "${NS_EMBED_ROOT}/index.html")
    message(FATAL_ERROR "Embedded asset tree needs index.html")
  endif()
  file(GLOB_RECURSE NS_EMBED_FILES CONFIGURE_DEPENDS LIST_DIRECTORIES FALSE "${NS_EMBED_ROOT}/*")
  set(NS_EMBED_CPP "${CMAKE_CURRENT_BINARY_DIR}/embedded_assets.cpp")
  file(WRITE "${NS_EMBED_CPP}.in" "// Generated trusted build-time asset snapshot; never edit.\n#include <neutronstar/host_window.hpp>\nstd::vector<neutronstar::EmbeddedAsset> neutronstar_embedded_assets() { return {\n")
  set(NS_EMBED_TOTAL 0)
  foreach(asset IN LISTS NS_EMBED_FILES)
    file(RELATIVE_PATH name "${NS_EMBED_ROOT}" "${asset}")
    if(IS_SYMLINK "${asset}" OR NOT name MATCHES "^[A-Za-z0-9_./-]+$")
      message(FATAL_ERROR "Invalid embedded asset path: ${name}")
    endif()
    file(SIZE "${asset}" length)
    math(EXPR NS_EMBED_TOTAL "${NS_EMBED_TOTAL}+${length}")
    if(length GREATER 33554432 OR NS_EMBED_TOTAL GREATER 134217728)
      message(FATAL_ERROR "Embedded asset size bound exceeded")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${asset}")
    file(READ "${asset}" hex HEX)
    string(REGEX REPLACE "(..)" "0x\\1," bytes "${hex}")
    file(APPEND "${NS_EMBED_CPP}.in" "{L\"${name}\", {${bytes}}},\n")
  endforeach()
  file(APPEND "${NS_EMBED_CPP}.in" "}; }\n")
  configure_file("${NS_EMBED_CPP}.in" "${NS_EMBED_CPP}" COPYONLY)
  add_executable(neutronstar_embedded_runner WIN32 examples/runner.cpp examples/app.rc "${NS_EMBED_CPP}")
  target_link_libraries(neutronstar_embedded_runner PRIVATE NeutronStar::Host shell32)
  target_include_directories(neutronstar_embedded_runner PRIVATE src)
  target_compile_definitions(neutronstar_embedded_runner PRIVATE NEUTRONSTAR_EMBEDDED_ASSETS)
  target_compile_options(neutronstar_embedded_runner PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /guard:cf)
  target_link_options(neutronstar_embedded_runner PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(neutronstar_policy_tests tests/policy_tests.cpp src/asset_policy.cpp src/test_greet_protocol.cpp)
  target_include_directories(neutronstar_policy_tests PRIVATE src include)
  target_compile_definitions(neutronstar_policy_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(neutronstar_policy_tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  add_test(NAME asset_policy COMMAND neutronstar_policy_tests)
endif()
